```
	CreatedAt int64  `gorm:"autoCreateTime:milli"`
	UpdatedAt int64  `gorm:"autoUpdateTime:milli"`
}

	// 1. 使用 filepath 替代 strings.Join 保证跨平台路径正确
	currentPath, _ := utils.GetCurrentPath()
	dbDir := filepath.Join(currentPath, "db")
	
	if _, err := os.Stat(dbDir); os.IsNotExist(err) {
		_ = os.MkdirAll(dbDir, 0755)
	}

	dbFile := filepath.Join(dbDir, "nas_core.db")

	// 2. 开启 GORM 配置优化
	var err error

	if err != nil {

	}

	// 3. 配置连接池
	sqlDB, _ := DB.DB()
```

### 1. 权限校验中间件的设计逻辑

为了实现“Go 控制权限”且“不交集”，中间件需要完成以下闭环：

1.  **身份识别**：从 Context 中获取 `UserID`（这步由 Auth 中间件完成）。
2.  **权限读取**：从 SQLite 中读出该用户的 `UserPath` 配置。
3.  **路径清洗**：防止 `..` 路径穿越。
4.  **范围比对**：判断请求的路径是否在授权的 JSON 列表中，并检查读写权限。

```
package middleware

import (
	"LiteNAS/database"
	"encoding/json"
	"net/http"
	"path/filepath"
	"strings"

	"github.com/gin-gonic/gin"
)

// UserPermission 对应数据库中的 JSON 结构
type UserPermission struct {
	Path  string `json:"path"`
	Label string `json:"label"`
	Read  bool   `json:"read"`
	Write bool   `json:"write"`
}

func FilePolicy() gin.HandlerFunc {
	return func(c *gin.Context) {
		// 1. 获取由 Auth 中间件存入的 UserID
		uidRaw, exists := c.Get("user_id")
		if !exists {
			c.JSON(http.StatusUnauthorized, gin.H{"status": 1, "message": "未授权"})
			c.Abort()
			return
		}
		uid := uidRaw.(uint)

		// 2. 获取请求的路径 (假设前端通过 query 或 post 传 path)
		reqPath := c.DefaultQuery("path", c.PostForm("path"))
		if reqPath == "" {
			c.Next() // 或者是返回错误，取决于你的接口设计
			return
		}

		// 3. 安全清洗：防止路径穿越 (最重要的防御)
		// Clean 会处理掉其中的 .. 或多余的 /
		safeReqPath := filepath.Clean(reqPath)

		// 4. 从数据库获取用户权限配置
		var user database.Manager
		if err := database.DB.Select("user_path", "is_admin").First(&user, uid).Error; err != nil {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "权限配置错误"})
			c.Abort()
			return
		}

		// 如果是超级管理员，直接放行所有路径
		if user.IsAdmin {
			c.Next()
			return
		}

		// 5. 解析权限 JSON
		var perms []UserPermission
		_ = json.Unmarshal([]byte(user.UserPath), &perms)

		// 6. 路径匹配校验
		authorized := false
		isWriteOp := isWriteMethod(c.Request.Method) // 判断是否是写操作

		for _, p := range perms {
			// 检查请求路径是否在授权路径之下
			// 注意：这里需要根据物理根目录进行逻辑判断
			if strings.HasPrefix(safeReqPath, filepath.Clean(p.Path)) {
				// 校验读写权限
				if isWriteOp && !p.Write {
					continue
				}
				if !isWriteOp && !p.Read {
					continue
				}
				authorized = true
				break
			}
		}

		if !authorized {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "无权访问该路径"})
			c.Abort()
			return
		}

		c.Next()
	}
}

// 辅助函数：判断是否为写操作
func isWriteMethod(method string) bool {
	m := strings.ToUpper(method)
	return m == "POST" || m == "PUT" || m == "DELETE" || m == "PATCH"
}

```

### 🌟 这种方案的优势与注意事项

1.  **路径清洗 (`filepath.Clean`)**： 绝对不能省。如果用户传 `path="/mnt/sda1/../../etc"`，`Clean` 后配合 `HasPrefix` 检查，可以有效防止用户跳出你分配给他的目录。
2.  **管理员豁免**： 代码中判断了 `IsAdmin`，这样作为 NAS 管理员的你，就不需要给自己配置复杂的路径 JSON，默认拥有全盘权限。
3.  **性能**： 由于是在路由组级别挂载，只有在操作文件时才会去查 SQLite 并解析 JSON。对于 NAS 这种操作频率（每秒几十次请求以内），这种开销微乎其微。

### 🛡️ 进阶构思：软连接问题

如果用户在 `/mnt/sda1` 下建立了一个指向 `/etc` 的软连接（Symbolic Link），单纯的路径字符串比对可能会被绕过。

*   **建议**：在主程序真正执行 `os.Open` 之前，可以使用 `filepath.EvalSymlinks` 拿到真实的物理路径，再做一次比对，这样安全性就达到了“企业级”。

### 1. 即使是 Admin 也要“画地为牢”

你的想法非常成熟。作为 NAS 的管理员，Web 界面应该是用来管理**数据**的，而不是用来破坏**系统分区**的。将系统级操作（如修改配置、关机）封装在特定的功能模块中，而文件管理插件只操作数据目录。

**优化方案：** 在主程序的全局配置文件（或常量）中定义一个 `DATA_ROOT`（例如 `/mnt`）。

*   所有的路径校验都必须以这个 `DATA_ROOT` 为基准。
*   即使是管理员，其授权路径也必须是 `DATA_ROOT` 的子集。

***

### 2. 软链接验证与“真实路径”获取

在 Go 中，处理软链接最核心的函数是 `filepath.EvalSymlinks(path)`。它的作用类似于执行 `readlink -f`：**它会追踪所有的软链接，直到找到磁盘上真实的物理文件或目录。**

**校验逻辑：**

1.  获取用户请求的路径 `reqPath`。
2.  使用 `filepath.EvalSymlinks(reqPath)` 拿到物理真实路径 `realPath`。
3.  检查 `realPath` 是否依然在用户被授权的目录范围内。

***

### 3. 强化版权限中间件实现

我们将这两个逻辑集成到中间件中，解决路径穿越和软链接逃逸。

#### 📂 middleware/file\_policy.go

```
package middleware

import (
	"LiteNAS/database"
	"encoding/json"
	"net/http"
	"path/filepath"
	"strings"

	"github.com/gin-gonic/gin"
)

// 系统强制的数据根目录，禁止任何 API 超出此范围
const GlobalDataRoot = "/mnt"

func FilePolicy() gin.HandlerFunc {
	return func(c *gin.Context) {
		uidRaw, _ := c.Get("user_id")
		uid := uidRaw.(uint)

		// 1. 获取请求路径
		reqPath := c.DefaultQuery("path", c.PostForm("path"))
		if reqPath == "" {
			c.JSON(http.StatusBadRequest, gin.H{"status": 1, "message": "缺少路径参数"})
			c.Abort()
			return
		}

		// 2. 深度解析：评估软链接并转为绝对路径
		// 这一步能把 ../../../etc/shadow 或者是指向系统目录的软链接还原成真实面目
		realPath, err := filepath.EvalSymlinks(filepath.Join(GlobalDataRoot, reqPath))
		if err != nil {
			// 如果路径不存在，可能无法评估软链接，这里可以根据业务决定是报错还是放行创建请求
			realPath = filepath.Clean(filepath.Join(GlobalDataRoot, reqPath))
		}

		// 3. 强制校验：是否超出了系统允许的数据根目录
		if !strings.HasPrefix(realPath, GlobalDataRoot) {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "非法操作：严禁访问系统目录"})
			c.Abort()
			return
		}

		// 4. 获取用户数据库权限
		var user database.Manager
		database.DB.Select("user_path", "is_admin").First(&user, uid)

		var perms []UserPermission
		_ = json.Unmarshal([]byte(user.UserPath), &perms)

		// 5. 权限比对
		authorized := false
		isWrite := isWriteMethod(c.Request.Method)

		for _, p := range perms {
			// 将授权路径也转为绝对物理路径进行比对
			authPath, _ := filepath.Abs(p.Path) 
			
			if strings.HasPrefix(realPath, authPath) {
				if isWrite && !p.Write { continue }
				if !isWrite && !p.Read { continue }
				authorized = true
				break
			}
		}

		// 特殊逻辑：Admin 如果没有在 UserPath 里配置，默认可以访问整个 GlobalDataRoot
		if user.IsAdmin && strings.HasPrefix(realPath, GlobalDataRoot) {
			authorized = true
		}

		if !authorized {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "不在授权访问范围内"})
			c.Abort()
			return
		}

		// 将校验后的“真实物理路径”存入上下文，方便后续 Controller 直接使用，无需再次解析
		c.Set("validated_path", realPath)
		c.Next()
	}
}

```

### 🌟 关键细节分析

1.  **`filepath.EvalSymlinks`**： 这是防范黑客的利器。如果用户在 `/mnt/data/` 下建了一个名为 `hack` 的软链接指向 `/etc/passwd`，当你访问 `/mnt/data/hack` 时，这个函数会返回 `/etc/passwd`。随后的 `strings.HasPrefix(realPath, GlobalDataRoot)` 就会发现它不再以 `/mnt` 开头，直接拦截。
2.  **`c.Set("validated_path", realPath)`**： 这是一个性能和安全的小技巧。在中间件校验完路径后，直接把物理路径传给下一个环节（Controller）。Controller 拿到的就是绝对安全的路径，直接用 `os.Open` 即可，不用再担心路径穿越问题。
3.  **Admin 的范围限制**： 通过 `GlobalDataRoot` 常量，我们给所有用户（包括 Admin）套上了一个“安全笼子”。Admin 的特殊性在于他不需要在数据库里逐个配置文件夹，但他依然不能跳出 `/mnt`。

### 🚀 避坑指南：

*   **路径拼接**：在 Linux 下，始终建议使用 `filepath.Join` 而不是简单的字符串相加，它会自动处理多余的斜杠。
*   **磁盘挂载点**：如果你的 `/mnt` 下挂载了不同的磁盘（如 `/mnt/sda1`, `/mnt/sdb1`），`EvalSymlinks` 依然有效。

```
const GlobalDataRoot = "/mnt"

// UserPermission 对应数据库中 Manager 表的 UserPath JSON 结构
type UserPermission struct {
	Path  string `json:"path"`
	Label string `json:"label"`
	Read  bool   `json:"read"`
	Write bool   `json:"write"`
}

// FilePolicy 文件操作权限校验中间件
func FilePolicy() gin.HandlerFunc {
	return func(c *gin.Context) {
		// 1. 获取由 Auth 中间件存入的 UserID
		uidRaw, exists := c.Get("user_id")
		if !exists {
			c.JSON(http.StatusUnauthorized, gin.H{"status": 1, "message": "未登录"})
			c.Abort()
			return
		}
		uid := uidRaw.(uint)

		// 2. 获取请求路径
		reqPath := c.DefaultQuery("path", c.PostForm("path"))
		if reqPath == "" {
			// 如果某些接口不需要路径参数也可以通过，可放行
			c.Next()
			return
		}

		// 3. 深度解析：评估软链接并转为绝对路径
		// filepath.Join 会处理掉相对路径，EvalSymlinks 会还原软链接的真实物理路径
		absPath := filepath.Join(GlobalDataRoot, reqPath)
		realPath, err := filepath.EvalSymlinks(absPath)
		if err != nil {
			// 如果路径不存在（比如用户正在创建新文件夹），EvalSymlinks 会报错
			// 此时我们回退到使用 Clean 保证路径格式正确
			realPath = filepath.Clean(absPath)
		}

		// 4. 强制校验：是否超出了系统允许的数据根目录（防止访问 /etc 等）
		if !strings.HasPrefix(realPath, GlobalDataRoot) {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "非法操作：严禁越权访问系统目录"})
			c.Abort()
			return
		}

		// 5. 获取用户数据库权限
		var user database.Manager
		if err := database.DB.Select("user_path", "is_admin").First(&user, uid).Error; err != nil {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "用户数据异常"})
			c.Abort()
			return
		}

		// 6. 权限比对逻辑
		authorized := false
		isWrite := isWriteMethod(c.Request.Method)

		// 管理员特权：只要在 GlobalDataRoot 范围内就放行
		if user.IsAdmin {
			authorized = true
		} else {
			// 普通用户解析 UserPath JSON
			var perms []UserPermission
			if err := json.Unmarshal([]byte(user.UserPath), &perms); err == nil {
				for _, p := range perms {
					// 将授权路径也转为清洁的绝对路径
					authPath := filepath.Clean(p.Path)
					// 检查请求路径是否在授权路径之下
					if strings.HasPrefix(realPath, authPath) {
						if isWrite && !p.Write {
							continue // 有读取权但无写入权，继续匹配其他可能的授权
						}
						if !isWrite && !p.Read {
							continue
						}
						authorized = true
						break
					}
				}
			}
		}

		if !authorized {
			c.JSON(http.StatusForbidden, gin.H{"status": 1, "message": "无访问权限或目录未授权"})
			c.Abort()
			return
		}

		// 7. 将校验后的“真实物理路径”存入 Context，Controller 直接使用 c.GetString("validated_path")
		c.Set("validated_path", realPath)
		c.Next()
	}
}

// isWriteMethod 辅助函数：判断是否为写操作请求
func isWriteMethod(method string) bool {
	m := strings.ToUpper(method)
	// POST(创建/修改), PUT(覆盖), DELETE(删除), PATCH(局部修改)
	return m == "POST" || m == "PUT" || m == "DELETE" || m == "PATCH"
}
```

```
// GenerateRandomKey 生成指定长度的随机字符串 (安全)
func GenerateRandomKey(n int) string {
	b := make([]byte, n)
	if _, err := rand.Read(b); err != nil {
		return ""
	}
	return hex.EncodeToString(b)[:n]
}

func CheckConfig(currentPath string) (*Config, error) {

		conf.AdminPWD = GenerateRandomKey(12)
		conf.FormMemory = 32
		
		data, _ := yaml.Marshal(conf)
		if err := os.WriteFile(configFile, data, 0644); err != nil {
```

```
	// 5. 初始化路由
	gin.SetMode(gin.DebugMode)
	app := router.InitRouter(conf.SECRET_KEY, currentPath, conf.FormMemory)

	srv := &http.Server{
		Addr:    ":" + conf.Port,
		Handler: app,
	}

	// 6. 优雅启动
	go func() {
		log.Printf("LiteNAS Core 运行在端口 %s\n", conf.Port)
		if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			log.Fatalf("服务启动失败: %v", err)
		}
	}()

	// 7. 信号监听 (支持 Interrupt 和 Terminate)
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, os.Interrupt, syscall.SIGTERM)
	<-quit

	log.Println("正在关闭服务...")

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	if err := srv.Shutdown(ctx); err != nil {
		log.Fatal("服务器强制关闭:", err)
	}

	log.Println("LiteNAS 已安全退出")
}
```

### 🌟 关键点解析：

1.  **`fmt.Sprintf` vs `strings.Join`**：对于简单的字符串拼接，`+` 或 `fmt.Sprintf` 可读性更好。
2.  **`crypto/rand`**：通过 `hex.EncodeToString` 生成的字符串直接就是 ASCII 字符，完全避开了之前 `math/rand` 需要 rune 数组映射的麻烦。
3.  **优雅退出升级**：增加了 `syscall.SIGTERM` 监听，这在未来你把 NAS 做成 Linux 服务或 Docker 容器时，能确保它接收到系统的停止指令并正确保存数据库。
4.  **`os.Getwd()`**：在大多数 Linux 环境下，它比 `os.Executable` 拿到的路径更符合“工作空间”的概念，但如果是作为系统服务运行，建议保留你原本的 `GetCurrentPath` 以确保找到 `db` 文件夹。
